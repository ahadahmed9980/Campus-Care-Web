Widget _buildUI(BuildContext context) {
  // Coin list se source initialize karein
  final CoinDataTableSource dataSource = CoinDataTableSource(coinList);

  return Container(
    width: double.infinity,
    padding: const EdgeInsets.all(16),
    child: PaginatedDataTable(
      header: const Text(
        "Coins Overview",
        style: TextStyle(fontWeight: FontWeight.bold),
      ),
      rowsPerPage: 5, // Ek page par 5 items dikhayega
      availableRowsPerPage: const [5, 10], // Dropdown options
      showCheckboxColumn: false, // Checkboxes hide karne ke liye
      columns: const [
        DataColumn(
          label: Text("ID", style: TextStyle(fontWeight: FontWeight.w900)),
        ),
        DataColumn(
          label: Text("Name", style: TextStyle(fontWeight: FontWeight.w900)),
        ),
        DataColumn(
          label: Text("Chain", style: TextStyle(fontWeight: FontWeight.w900)),
        ),
        DataColumn(
          label: Text("Price", style: TextStyle(fontWeight: FontWeight.w900)),
        ),
      ],
      source: dataSource,
    ),
  );
}
class CoinDataTableSource extends DataTableSource {
  final List<Coin> data;

  CoinDataTableSource(this.data);

  @override
  DataRow? getRow(int index) {
    if (index >= data.length) return null;
    final coin = data[index];
    return DataRow.byIndex(
      index: index,
      cells: [
        DataCell(Text(coin.id.toString())),
        DataCell(Text(coin.name)),
        DataCell(Text(coin.chainName)),
        DataCell(Text("\$${coin.price.toStringAsFixed(2)}")),
      ],
    );
  }

  @override
  bool get isRowCountApproximate => false;

  @override
  int get rowCount => data.length;

  @override
  int get selectedRowCount => 0;
}
